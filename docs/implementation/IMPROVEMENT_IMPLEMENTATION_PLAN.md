from django.db import transaction

@transaction.atomic
def bulk_approve(request):
    """موافقة جماعية محسنة"""
    ids = request.POST.getlist('ids[]')
    
    # معالجة دفعات من 50
    for i in range(0, len(ids), 50):
        batch = ids[i:i+50]
        Entry.objects.filter(id__in=batch).update(
            status='approved',
            approved_by=request.user,
            approved_at=timezone.now()
        )
    
    return JsonResponse({'success': True, 'count': len(ids)})

class ProductionProgressTracker:
    def get_visual_progress(self, order):
        stages = order.stages.all()
        completed = stages.filter(status='completed').count()
        total = stages.count()
        
        return {
            'percentage': (completed / total * 100) if total > 0 else 0,
            'completed': completed,
            'total': total,
            'current_stage': stages.filter(status='in_progress').first(),
            'is_delayed': self.check_delay(order)
        }
